# SM_INIT=badcrc.bin: PROG.BAS with a changed byte. LOAD reports the CRC; the program stays as it was read.
wait 1.5
type LOAD "PROG"\n
waitfor 1 SUPER MEMORIA: INTRO 5
cart sm
press INTRO
waitfor 0 Done. 5
cart rom
press INTRO
wait 1.2
waitfor 0 Bad CRC 2
press AVANCE
wait 0.3
type LIST\n
wait 0.6
expectscr 2    10 PLOT "hi"

quit
